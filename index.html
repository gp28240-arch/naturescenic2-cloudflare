async function sendTranslationRequest() { const text = document.getElementById('trans-text-input').value.trim(); const source = document.getElementById('trans-source-lang').value; const target = document.getElementById('trans-target-lang').value; const resBox = document.getElementById('translation-result'); if (!text) { alert("لطفاً متنی برای ترجمه وارد کنید."); return; } resBox.innerText = "در حال ترجمه... لطفاً منتظر بمانید."; try { const url = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=${source}&tl=${target}&dt=t&q=${encodeURIComponent(text)}`; const response = await fetch(url); if (!response.ok) throw new Error("خطا در شبکه"); const data = await response.json(); let translatedText = ''; if (data && data[0]) { data[0].forEach(item => { if (item[0]) translatedText += item[0]; }); } if (translatedText) { resBox.innerText = translatedText; } else { resBox.innerText = "ترجمه‌ای یافت نشد."; } } catch (e) { console.error("خطای ترجمه:", e); resBox.innerText = "خطا در دریافت ترجمه. ممکن است دسترسی اینترنت یا سرویس ترجمه محدود شده باشد."; } } function playTextToSpeech() { const text = document.getElementById('translation-result').innerText; const targetLang = document.getElementById('trans-target-lang').value; if (!text || text === "نتیجه ترجمه اینجا نمایش داده می‌شود..." || text.startsWith("در حال ترجمه") || text.startsWith("خطا")) { alert("متنی برای خواندن وجود ندارد."); return; } // ۱. اولویت اول: استفاده از سرویس صوتی گوگل (قابل اجرا در اپ موبایل و وب) try { const audioUrl = `https://translate.google.com/translate_tts?ie=UTF-8&q=${encodeURIComponent(text)}&tl=${targetLang}&client=tw-ob`; const audio = new Audio(audioUrl); audio.play().catch(() => { // ۲. اولویت دوم: پشتیبان مرورگر در صورت مسدود شدن پخش خودکار fallbackWebSpeech(text, targetLang); }); } catch (e) { fallbackWebSpeech(text, targetLang); } } function fallbackWebSpeech(text, lang) { if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(text); utterance.lang = lang; window.speechSynthesis.speak(utterance); } else { alert("قابلیت خواندن صوتی در این دستگاه یا مرورگر پشتیبانی نمی‌شود."); } }
